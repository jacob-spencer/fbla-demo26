Database Engine: MySQL
Character Set: UTF8

TABLE: sessions
--------------
sid VARCHAR(255) PRIMARY KEY
sess JSONB NOT NULL
expire TIMESTAMP NOT NULL
INDEX: IDX_session_expire ON expire

TABLE: users
-----------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
username VARCHAR(50) NOT NULL UNIQUE
email VARCHAR(255) NOT NULL UNIQUE
password VARCHAR(255) NOT NULL
first_name VARCHAR(255)
last_name VARCHAR(255)
profile_image_url VARCHAR(255)
bio TEXT
location VARCHAR(255)
is_business_owner BOOLEAN DEFAULT false
created_at TIMESTAMP DEFAULT NOW()
updated_at TIMESTAMP DEFAULT NOW()

TABLE: categories
----------------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
name VARCHAR(100) NOT NULL
icon VARCHAR(50) NOT NULL
slug VARCHAR(100) NOT NULL UNIQUE
created_at TIMESTAMP DEFAULT NOW()

TABLE: businesses
----------------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
name VARCHAR(255) NOT NULL
description TEXT
category_id VARCHAR(255) NOT NULL REFERENCES categories(id)
owner_id VARCHAR(255) REFERENCES users(id)
address TEXT NOT NULL
phone VARCHAR(20)
email VARCHAR(255)
website VARCHAR(255)
image_url VARCHAR(255)
banner_image_url VARCHAR(255)
latitude REAL
longitude REAL
price_range VARCHAR(10) DEFAULT '$$'
is_open BOOLEAN DEFAULT true
opening_hours JSONB
average_rating REAL DEFAULT 0
total_reviews INTEGER DEFAULT 0
created_at TIMESTAMP DEFAULT NOW()
updated_at TIMESTAMP DEFAULT NOW()

TABLE: deals
-----------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
business_id VARCHAR(255) NOT NULL REFERENCES businesses(id)
title VARCHAR(255) NOT NULL
description TEXT NOT NULL
discount_type VARCHAR(20) NOT NULL
discount_value VARCHAR(50) NOT NULL
discount_code VARCHAR(40) NOT NULL
expires_at TIMESTAMP NOT NULL
is_active BOOLEAN DEFAULT true
created_at TIMESTAMP DEFAULT NOW()

TABLE: reviews
-------------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
business_id VARCHAR(255) NOT NULL REFERENCES businesses(id)
user_id VARCHAR(255) NOT NULL REFERENCES users(id)
rating INTEGER NOT NULL
comment TEXT
is_verified BOOLEAN DEFAULT false
verification_email VARCHAR(255)
verification_code VARCHAR(255)
verification_expires_at TIMESTAMP
created_at TIMESTAMP DEFAULT NOW()

TABLE: review_replies
--------------------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
review_id VARCHAR(255) NOT NULL REFERENCES reviews(id)
user_id VARCHAR(255) NOT NULL REFERENCES users(id)
reply TEXT NOT NULL
is_business_owner BOOLEAN DEFAULT false
created_at TIMESTAMP DEFAULT NOW()

TABLE: bookmarks
---------------
id VARCHAR(255) PRIMARY KEY DEFAULT gen_random_uuid()
user_id VARCHAR(255) NOT NULL REFERENCES users(id)
business_id VARCHAR(255) NOT NULL REFERENCES businesses(id)
created_at TIMESTAMP DEFAULT NOW()

TABLE: analytics
---------------
id CHAR(36) PRIMARY KEY DEFAULT gen_random_uuid()
business_id CHAR(36)
event_type VARCHAR(100)
event_timestamp TIMESTAMP current_timestamp()
session_id VARCHAR(255)
user_id VARCHAR(255)
page_url TEXT
referrer_url TEXT
device_type VARCHAR(50)
browser VARCHAR(100)
os VARCHAR(100)
country VARCHAR(100)
city VARCHAR(100)
metadata LONGTEXT

FOREIGN KEY RELATIONSHIPS:
=========================
- businesses.category_id → categories.id
- businesses.owner_id → users.id
- deals.business_id → businesses.id
- reviews.business_id → businesses.id
- reviews.user_id → users.id
- review_replies.review_id → reviews.id
- review_replies.user_id → users.id
- bookmarks.user_id → users.id
- bookmarks.business_id → businesses.id

INDEXES:
=======
- sessions.expire (IDX_session_expire)
- users.username (UNIQUE)
- users.email (UNIQUE)
- categories.slug (UNIQUE)
